# Build Docker image
    docker build -t my-go-app .

# Run Docker container
    docker run -d -p 8080:8080 --name docker-go-app my-go-app

# Test
    Mở trình duyệt và truy cập http://localhost:8080. Bạn sẽ thấy dòng chữ "Hello, Docker Go!".
    Hoặc dùng curl: curl http://localhost:8080

# Xem Logs
    docker logs docker-go-app

# Stop Docker container
    docker stop docker-go-app

# Remove Docker container
    docker rm docker-go-app
