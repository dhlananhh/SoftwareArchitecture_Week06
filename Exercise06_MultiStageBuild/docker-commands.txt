# Build Docker image
    docker build -t my-node-multistage-app .

# Run Docker container
    docker run -d -p 3001:3000 --name node-multi my-node-multistage-app
    # Dùng cổng 3001 trên host để tránh trùng với Bài 1 nếu còn chạy

# Test
    Mở trình duyệt hoặc dùng curl: http://localhost:3001
    So sánh kích thước image này (docker images my-node-multistage-app) với image không dùng multi-stage (nếu có) để thấy sự khác biệt.

# Stop Docker container
    docker stop node-multi

# Remove Docker container
    docker rm node-multi
