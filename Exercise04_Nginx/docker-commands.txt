# Build Docker image
docker build -t my-static-web .

# Run Docker container
docker run -d -p 8080:80 --name static-nginx my-static-web

# Test
Mở trình duyệt và truy cập http://localhost:8080. Bạn sẽ thấy nội dung file index.html.

# Stop Docker container
docker stop static-nginx

# Remove Docker container
docker rm static-nginx
