# Build the image
    docker build -t my-custom-postgres .

# Run the container
    docker run -d \
        --name mypostgres \
        -e POSTGRES_PASSWORD=postgres \
        -p 5432:5432 \
        -v pgdata:/var/lib/postgresql/data \
        my-custom-postgres

# Xem Logs
    docker logs mypostgres

# Test the connection
    docker exec -it postgres psql -U postgres

# Test the container
psql -h localhost -U postgres -d postgres -c "SELECT * FROM test;"

# Stop Docker connection
    docker stop mypostgres

# Remove Docker container
    docker rm mypostgres
