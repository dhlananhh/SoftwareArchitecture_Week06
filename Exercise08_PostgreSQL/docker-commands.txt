# Build the image
    docker build -t my-custom-postgres .

# Run the container
    docker run -d \
        --name postgres \
        -e POSTGRES_PASSWORD=postgres \
        -p 5432:5432 \
        -v pgdata:/var/lib/postgresql/data \
        my-custom-postgres
    
    docker run -d --name postgres -e POSTGRES_PASSWORD=postgres -p 5432:5432 -v pgdata:/var/lib/postgresql/data my-custom-postgres

    docker run -d --name postgres -e POSTGRES_PASSWORD=postgres -p 5433:5432 -v pgdata:/var/lib/postgresql/data my-custom-postgres

# Xem Logs
    docker logs my-custom-postgres

# Test the connection
    docker exec -it postgres psql -U postgres

# Test the container
    psql -h localhost -p 5433 -U postgres -d mydb -W
or:
    docker exec -it postgres psql -U postgres -d mydb -W

# Stop Docker connection
    docker stop my-custom-postgres

# Remove Docker container
    docker rm my-custom-postgres
