# Build Docker image
    docker build -t my-react-app .

# Run Docker image
    docker run -d -p 80:3000 --name react-prod-app my-react-app

# Test
    Mở trình duyệt và truy cập http://localhost (hoặc http://localhost:80 nếu bạn ánh xạ tới cổng 80). Bạn sẽ thấy ứng dụng React của mình.
    curl http://localhost:80

# Stop Docker container
    docker stop react-prod-app

# Remove Docker container
    docker rm react-prod-app
